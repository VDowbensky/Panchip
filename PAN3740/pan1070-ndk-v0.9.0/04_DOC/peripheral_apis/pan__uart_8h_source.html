118/*
119 * DESCRIPTION
120 * This is the data type used for specifying UART events. An event is
121 * the occurrence of an interrupt which must be handled appropriately.
122 * One of these events is passed at a time to the user listener
123 * function to be processed. The exception to this are the
124 * UART_EVENT_THR_EMPTY and UART_EVENT_TIMEOUT interrupts which are handled
125 * internally by the interrupt handler.
126 */
127typedef enum UART_Event
128{
129 UART_EVENT_MODEM = 0x0, // CTR, DSR, RI or DCD status changed.
130 UART_EVENT_NONE = 0x1, // No event/interrupt pending.
131 UART_EVENT_THR_EMPTY = 0x2, // Transmit holding register empty or TX
132 // FIFO at or below trigger threshold.
133 UART_EVENT_DATA = 0x4, // Receive buffer register data
134 // available (non-FIFO mode) or RX FIFO
135 // trigger level reached.
136 UART_EVENT_LINE = 0x6, // Overrun/parity/framing error or break
137 // interrupt occurred.
138#if 0
139 UART_EVENT_BUSY = 0x7, // Attempt to write to the LCR[7] while
140 // DW_apb_uart was busy (DLAB).
141#endif
142 UART_EVENT_TIMEOUT = 0xc // No characters in or out of the
143 // receiver FIFO during the last four
144 // character times and there is at least
145 // one character in it during this time.
146} UART_EventDef;
